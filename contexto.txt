import { useEffect, useRef, useState } from 'react'

const idSeccion = (categoria) => `categoria-${categoria.id}`

export default function NavCategorias({ categorias }) {
  const [activa, setActiva] = useState(categorias[0]?.id)
  const contenedor = useRef(null)

  // Marca como activa la primera categoría que toca la franja ubicada
  // justo debajo de la barra fija.
  useEffect(() => {
    const visibles = new Set()

    const elegir = () => {
      const alFinal =
        window.innerHeight + window.scrollY >= document.documentElement.scrollHeight - 2
      if (alFinal) {
        setActiva(categorias[categorias.length - 1].id)
        return
      }
      const primera = categorias.find((c) => visibles.has(c.id))
      if (primera) setActiva(primera.id)
    }

    const observer = new IntersectionObserver(
      (entradas) => {
        entradas.forEach((entrada) => {
          const id = entrada.target.dataset.categoria
          if (entrada.isIntersecting) visibles.add(id)
          else visibles.delete(id)
        })
        elegir()
      },
      { rootMargin: '-64px 0px -60% 0px' },
    )

    categorias.forEach((categoria) => {
      const seccion = document.getElementById(idSeccion(categoria))?.closest('section')
      if (!seccion) return
      seccion.dataset.categoria = categoria.id
      observer.observe(seccion)
    })

    window.addEventListener('scroll', elegir, { passive: true })
    return () => {
      observer.disconnect()
      window.removeEventListener('scroll', elegir)
    }
  }, [categorias])

  // En celular, mantiene visible el enlace activo dentro de la barra.
  useEffect(() => {
    const barra = contenedor.current
    const enlace = barra?.querySelector('[aria-current]')
    if (!enlace) return
    const sinMovimiento = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    barra.scrollTo({
      left: enlace.offsetLeft - (barra.clientWidth - enlace.offsetWidth) / 2,
      behavior: sinMovimiento ? 'auto' : 'smooth',
    })
  }, [activa])

  return (
    <nav className="nav" aria-label="Categorías">
      <div className="nav__barra contenedor" ref={contenedor}>
        <ul>
          {categorias.map((categoria) => (
            <li key={categoria.id}>
              <a
                href={`#${idSeccion(categoria)}`}
                aria-current={activa === categoria.id ? 'true' : undefined}
              >
                {categoria.nombre}
              </a>
            </li>
          ))}
        </ul>
      </div>
    </nav>
  )
}

// Única puerta de acceso a los datos de los locales.
// Hoy lee archivos JSON del proyecto; más adelante se puede reemplazar por
// una base de datos sin tocar el resto de la app, mientras getLocal siga
// devolviendo la misma forma de datos (o null si el local no existe).

const archivos = import.meta.glob('../data/locales/*.json', {
  eager: true,
  import: 'default',
})

const locales = Object.fromEntries(
  Object.values(archivos).map((local) => [local.slug, local]),
)

// Las imágenes se guardan en el JSON solo con el nombre del archivo;
// acá se resuelve la ruta completa dentro de public/locales/<slug>/.
function resolverImagen(slug, imagen) {
  return imagen ? `/locales/${slug}/${imagen}` : null
}

function normalizar(local) {
  return {
    ...local,
    logo: resolverImagen(local.slug, local.logo),
    categorias: local.categorias.map((categoria) => ({
      ...categoria,
      productos: categoria.productos.map((producto) => ({
        ...producto,
        imagen: resolverImagen(local.slug, producto.imagen),
      })),
    })),
  }
}

export async function getLocal(slug) {
  const local = locales[slug]
  return local ? normalizar(local) : null
}

=== src\lib\useAparecer.js ===
import { useEffect, useRef, useState } from 'react'

// Devuelve [ref, visible]: visible pasa a true la primera vez que el elemento entra en pantalla.
export function useAparecer() {
  const ref = useRef(null)
  const [visible, setVisible] = useState(() => !('IntersectionObserver' in window))

  useEffect(() => {
    const elemento = ref.current
    if (visible || !elemento) return
    const observer = new IntersectionObserver(
      ([entrada]) => {
        if (entrada.isIntersecting) {
          setVisible(true)
          observer.disconnect()
        }
      },
      { threshold: 0.05 },
    )
    observer.observe(elemento)
    return () => observer.disconnect()
  }, [visible])

  return [ref, visible]
}

=== src\lib\useLocal.js ===
import { useEffect, useState } from 'react'
import { getLocal } from './locales'

// Carga el local del slug indicado. Devuelve { cargando, local }:
// local es null si el slug no existe.
export function useLocal(slug) {
  // Guardamos el slug junto con el resultado para saber si corresponde al actual.
  const [estado, setEstado] = useState({ slug: null, local: null })

  useEffect(() => {
    let vigente = true
    getLocal(slug).then((local) => {
      if (vigente) setEstado({ slug, local })
    })
    return () => {
      vigente = false
    }
  }, [slug])

  return { cargando: estado.slug !== slug, local: estado.local }
}

import { useEffect, useState } from 'react'
import { useParams } from 'react-router-dom'
import { useLocal } from '../lib/useLocal'
import { variablesTema } from '../lib/tema'
import { urlPublicaMenu } from '../lib/urlPublica'
import NoEncontrado from './NoEncontrado'

const OPCIONES_QR = { errorCorrectionLevel: 'M', margin: 4 }

function descargar(href, nombreArchivo) {
  const enlace = document.createElement('a')
  enlace.href = href
  enlace.download = nombreArchivo
  enlace.click()
}

export default function QrPage() {
  const { slug } = useParams()
  const { cargando, local } = useLocal(slug)
  const [qr, setQr] = useState({ svg: null, error: false })
  const url = urlPublicaMenu(slug)

  useEffect(() => {
    if (!local) return
    let vigente = true
    // La librería se carga solo en esta página para no engordar el menú.
    import('qrcode')
      .then((QRCode) => QRCode.toString(url, { ...OPCIONES_QR, type: 'svg' }))
      .then((svg) => vigente && setQr({ svg, error: false }))
      .catch(() => vigente && setQr({ svg: null, error: true }))
    return () => {
      vigente = false
    }
  }, [local, url])

  useEffect(() => {
    if (local) document.title = `QR · ${local.nombre}`
  }, [local])

  if (cargando) return <p>Cargando…</p>
  if (!local) return <NoEncontrado />

  const descargarSvg = () => {
    const blob = new Blob([qr.svg], { type: 'image/svg+xml' })
    const href = URL.createObjectURL(blob)
    descargar(href, `qr-${slug}.svg`)
    URL.revokeObjectURL(href)
  }

  // Alta resolución (1200 px) para que se pueda imprimir sin perder nitidez.
  const descargarPng = async () => {
    const QRCode = await import('qrcode')
    const href = await QRCode.toDataURL(url, { ...OPCIONES_QR, width: 1200 })
    descargar(href, `qr-${slug}.png`)
  }

  return (
    <div className="menu qr" style={variablesTema(local.tema)}>
      <main className="contenedor">
        <section className="qr__tarjeta" aria-labelledby="qr-titulo">
          <h1 id="qr-titulo">{local.nombre}</h1>

          {qr.svg && (
            <div
              className="qr__codigo"
              role="img"
              aria-label={`Código QR del menú de ${local.nombre}`}
              dangerouslySetInnerHTML={{ __html: qr.svg }}
            />
          )}
          {qr.error && <p role="alert">No pudimos generar el código QR. Probá recargar la página.</p>}

          <p className="qr__texto">Escaneá para ver el menú</p>
          <p className="qr__url">{url}</p>
        </section>

        <div className="qr__acciones">
          <button type="button" onClick={descargarPng} disabled={!qr.svg}>
            Descargar PNG
          </button>
          <button type="button" onClick={descargarSvg} disabled={!qr.svg}>
            Descargar SVG
          </button>
        </div>
      </main>
    </div>
  )
}
